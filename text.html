async function startHostedCheckout() {
    // 1. طلب إنشاء الجلسة من سيرفر .NET
    const response = await fetch('/api/payments/create-hosted-session', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            productName: "اشتراك خطة برو سنوية",
            amount: 120.00
        })
    });

    const data = await response.json();

    // 2. التوجيه المباشر لرابط صفحة الدفع الخارجية
    if (data.redirectUrl) {
        window.location.href = data.redirectUrl;
    }
}

[HttpPost("create-hosted-session")]
public async Task<IActionResult> CreateHostedSession([FromBody] HostedCheckoutDto dto)
{
    var options = new SessionCreateOptions
    {
        PaymentMethodTypes = new List<string> { "card" },
        LineItems = new List<SessionLineItemOptions>
        {
            new SessionLineItemOptions
            {
                PriceData = new SessionLineItemPriceDataOptions
                {
                    UnitAmount = (long)(dto.Amount * 100), // المبلغ بالسنتاً
                    Currency = "usd",
                    ProductData = new SessionLineItemPriceDataProductDataOptions { Name = dto.ProductName }
                },
                Quantity = 1
            }
        },
        Mode = "payment",
        SuccessUrl = "https://yourdomain.com/payment-success?session_id={CHECKOUT_SESSION_ID}",
        CancelUrl = "https://yourdomain.com/payment-cancelled"
    };

    var service = new SessionService();
    Session session = await service.CreateAsync(options);

    return Ok(new { redirectUrl = session.Url });
}

الربط المباشر (Direct API)

إرسال بيانات البطاقة مباشرة لسيرفرك

PCI-DSS High Risk
تنبيه أمني هام: هذه الطريقة تتطلب الحصول على شهادة امتثال PCI-DSS Level 1 لأن سيرفرك يمر عليه بيانات البطاقة بشكل غير مشفر من البوابة.

كود الـ Backend (.NET Direct Charge API) 

[HttpPost("charge-direct")]
public async Task<IActionResult> ChargeDirect([FromBody] DirectCardPaymentDto dto)
{
    // 1. إنشاء PaymentMethod يدوي من بيانات البطاقة الخام
    var pmOptions = new PaymentMethodCreateOptions
    {
        Type = "card",
        Card = new PaymentMethodCardOptions
        {
            Number = dto.CardNumber,
            ExpMonth = dto.ExpMonth,
            ExpYear = dto.ExpYear,
            Cvc = dto.Cvc
        }
    };

    var pmService = new PaymentMethodService();
    PaymentMethod paymentMethod = await pmService.CreateAsync(pmOptions);

    // 2. إرسال طلب الخصم المباشر
    var intentOptions = new PaymentIntentCreateOptions
    {
        Amount = (long)(dto.Amount * 100),
        Currency = "usd",
        PaymentMethod = paymentMethod.Id,
        Confirm = true,
        OffSession = true
    };

    var intentService = new PaymentIntentService();
    PaymentIntent intent = await intentService.CreateAsync(intentOptions);

    return Ok(new { status = intent.Status, transactionId = intent.Id });
}

[HttpPost("webhook")]
public async Task<IActionResult> StripeWebhook()
{
    var json = await new StreamReader(HttpContext.Request.Body).ReadToEndAsync();
    var stripeSignature = Request.Headers["Stripe-Signature"];
    var webhookSecret = _config["Stripe:WebhookSecret"];

    try
    {
        // 1. التحقق التلقائي من التوقيع والوقت لمنع التزوير Replay Attacks
        var stripeEvent = EventUtility.ConstructEvent(json, stripeSignature, webhookSecret);

        // 2. معالجة الحدث الموثوق
        if (stripeEvent.Type == Events.CheckoutSessionCompleted)
        {
            var session = stripeEvent.Data.Object as Stripe.Checkout.Session;
            
            // تحديث حالة الطلب في قاعدة البيانات إلى مدفوع
            await UpdateOrderStatus(session.Id, OrderStatus.Paid);
        }

        return Ok();
    }
    catch (StripeException ex)
    {
        return BadRequest(new { error = ex.Message });
    }
}
